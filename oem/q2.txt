SELECT 
    inst.instance_name,
    tsu.tablespace_name, 
    ROUND(tsu.used_space/(1024*1024),2) AS used_mb, 
    ROUND(tsu.tablespace_size/(1024*1024),2) AS total_mb, 
    ROUND(tsu.used_percent, 2) AS used_percent
FROM 
    dba_tablespace_usage_metrics tsu
CROSS JOIN 
    v$instance inst
WHERE 
    ROUND(tsu.used_percent,2) > 85;
