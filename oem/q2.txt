SELECT
  t.target_name         AS "Database Name",
  ts.tablespace_name    AS "Tablespace Name",
  ROUND((ts.tablespace_used_size/ts.tablespace_size)*100,2) AS "Percent Used",
  ROUND(ts.tablespace_used_size/1024/1024,2) AS "Used MB",
  ROUND(ts.tablespace_size/1024/1024,2) AS "Total MB"
FROM
  sysman.mgmt_targets t,
  sysman.mgmt$db_tablespaces ts
WHERE
  t.target_guid = ts.target_guid
  AND t.target_type = 'oracle_database'
  AND ROUND((ts.tablespace_used_size/ts.tablespace_size)*100,2) > 85
ORDER BY t.target_name, "Percent Used" DESC;
