SELECT
  db.target_name            AS "Database Name",
  m.tablespace_name         AS "Tablespace Name",
  ROUND(m.used_percent,2)   AS "Percent Used",
  ROUND(m.used_space*8/1024,2)  AS "Used MB",
  ROUND(m.tablespace_size*8/1024,2) AS "Total MB"
FROM
  sysman.mgmt_targets db
JOIN
  sysman.mgmt$db_tablespace m
    ON m.target_guid = db.target_guid
JOIN
  sysman.mgmt$group_members gm
    ON gm.MEMBER_TARGET_GUID = db.target_guid
JOIN
  sysman.mgmt_targets grp
    ON gm.GROUP_TARGET_GUID = grp.target_guid
WHERE
  grp.target_type = 'oracle_group'
  AND grp.target_name = 'DB_Monitoring'
  AND m.used_percent > 85
ORDER BY db.target_name, "Percent Used" DESC
